  <script>
    const ADMIN_WEBHOOK_URL = "https://discord.com/api/webhooks/1549806286554796125/KizdckOtzhJS7VY4WriofF9sR-Eei11q-BRvPtBolP-QuyogFMnC-Dj3KkoP_PJPir6S";

    const PRODUCTS = {
      "Fortnite Private": { "3 Days": 7.99, "1 Week": 13.99, "1 Month": 29.99 },
      "Fortnite Public": { "3 Days": 4.99, "1 Week": 8.99, "1 Month": 18.99 },
      "Temp Spoofer": { "3 Days": 6.99, "1 Week": 12.99, "1 Month": 25.99 },
      "IPVanish": { "1 Week": 0.50, "1 Month": 2.99 }
    };

    const WALLETS = {
      "Litecoin (LTC)": "LNPT3jNWXfYeXr1WHgp8RoBzUBp1SUftYB",
      "Bitcoin (BTC)": "bc1q4j3vk8w0y96rwqyjrhlpylzm2uq37w25yly026",
      "Ethereum (ETH)": "0xA7f7467Ef4b58AC7910391d8B465f1cEB1b2F401"
    };

    let appState = {
      users: JSON.parse(localStorage.getItem('store_users') || '[]'),
      orders: JSON.parse(localStorage.getItem('store_orders') || '{}'),
      tickets: JSON.parse(localStorage.getItem('store_tickets') || '{}')
    };

    let currentUserEmail = null;
    let activeInvoiceId = null;
    let pollInterval = null;

    function saveState() {
      localStorage.setItem('store_users', JSON.stringify(appState.users));
      localStorage.setItem('store_orders', JSON.stringify(appState.orders));
      localStorage.setItem('store_tickets', JSON.stringify(appState.tickets));
    }

    async function sendWebhook(title, description, fields = []) {
      try {
        await fetch(ADMIN_WEBHOOK_URL, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            username: "Store System",
            embeds: [{ title, description, fields, timestamp: new Date().toISOString() }]
          })
        });
      } catch (err) {
        console.error("Webhook error:", err);
      }
    }

    function showView(viewName) {
      document.querySelectorAll('.view-section').forEach(el => el.classList.remove('active'));
      if (!currentUserEmail && (viewName === 'shop' || viewName === 'payment')) {
        document.getElementById('view-login').classList.add('active');
        return;
      }
      document.getElementById(`view-${viewName}`).classList.add('active');
      if (viewName === 'admin-dashboard') renderAdminDashboard();
    }

    document.getElementById('login-form').addEventListener('submit', (e) => {
      e.preventDefault();
      const email = document.getElementById('login-email').value;
      if (!appState.users.includes(email)) {
        appState.users.push(email);
        saveState();
      }
      currentUserEmail = email;
      document.getElementById('user-display').innerText = email;
      document.getElementById('ticket-email').value = email;
      initShop();
      showView('shop');
    });

    function initShop() {
      const prodSelect = document.getElementById('product-select');
      prodSelect.innerHTML = '';
      Object.keys(PRODUCTS).forEach(prod => {
        const opt = document.createElement('option');
        opt.value = prod;
        opt.innerText = prod;
        prodSelect.appendChild(opt);
      });
      updateOptions();
    }

    function updateOptions() {
      const prod = document.getElementById('product-select').value;
      const optSelect = document.getElementById('option-select');
      optSelect.innerHTML = '';
      Object.keys(PRODUCTS[prod]).forEach(optKey => {
        const opt = document.createElement('option');
        opt.value = optKey;
        opt.innerText = `${optKey} - $${PRODUCTS[prod][optKey].toFixed(2)}`;
        optSelect.appendChild(opt);
      });
      calculatePrice();
    }

    function calculatePrice() {
      const prod = document.getElementById('product-select'].value;
      const opt = document.getElementById('option-select').value;
      const basePrice = PRODUCTS[prod][opt] || 0;
      const tip = parseFloat(document.getElementById('tip-input').value) || 0;
      const total = basePrice + tip;

      document.getElementById('price-base').innerText = `$${basePrice.toFixed(2)}`;
      document.getElementById('price-tip').innerText = `$${tip.toFixed(2)}`;
      document.getElementById('price-total').innerText = `$${total.toFixed(2)}`;
    }

    async function submitOrder() {
      const prod = document.getElementById('product-select').value;
      const opt = document.getElementById('option-select').value;
      const basePrice = PRODUCTS[prod][opt];
      const tip = parseFloat(document.getElementById('tip-input').value) || 0;
      const totalPrice = parseFloat((basePrice + tip).toFixed(2));
      const cryptoType = document.getElementById('crypto-select').value;

      const invoiceId = 'INV-' + Math.random().toString(36).substring(2, 8).toUpperCase();
      
      const order = {
        invoiceId,
        email: currentUserEmail,
        product: prod,
        option: opt,
        totalPrice,
        cryptoType,
        cryptoAddress: WALLETS[cryptoType],
        cryptoAmount: (totalPrice / 85).toFixed(4),
        status: 'PENDING',
        remainingPrice: 0
      };

      appState.orders[invoiceId] = order;
      saveState();
      activeInvoiceId = invoiceId;

      renderPaymentScreen(order);
      showView('payment');
      startPolling();

      await sendWebhook("🚨 New Order", `Invoice: ${invoiceId}`, [
        { name: "Customer", value: currentUserEmail, inline: true },
        { name: "Total", value: `$${totalPrice.toFixed(2)}`, inline: true }
      ]);
    }

    function renderPaymentScreen(order) {
      document.getElementById('pay-invoice-id').innerText = order.invoiceId;
      document.getElementById('pay-product').innerText = order.product;
      document.getElementById('pay-option').innerText = order.option;
      document.getElementById('pay-total-usd').innerText = `$${order.totalPrice.toFixed(2)}`;
      document.getElementById('pay-total-crypto').innerText = `${order.cryptoAmount} ${order.cryptoType.split(' ')[1]}`;
      document.getElementById('pay-address').value = order.cryptoAddress;

      const badge = document.getElementById('payment-status-badge');
      const dcBtn = document.getElementById('dc-redirect-btn');
      const underpaidBox = document.getElementById('underpaid-box');

      badge.innerText = order.status;
      if (order.status === 'PAID') {
        badge.className = 'status-badge status-paid';
        dcBtn.style.display = 'block';
        underpaidBox.style.display = 'none';
      } else if (order.status === 'PARTIAL') {
        badge.className = 'status-badge status-partial';
        dcBtn.style.display = 'none';
        underpaidBox.style.display = 'block';
        document.getElementById('underpaid-amount').innerText = `$${order.remainingPrice.toFixed(2)} fehlen`;
      } else {
        badge.className = 'status-badge status-pending pulse';
        dcBtn.style.display = 'none';
        underpaidBox.style.display = 'none';
      }
    }

    // Automatischer Polling-Checker, der jetzt die lokalen Daten aktualisiert
    function startPolling() {
      if (pollInterval) clearInterval(pollInterval);
      pollInterval = setInterval(() => {
        if (!activeInvoiceId) return;
        appState.orders = JSON.parse(localStorage.getItem('store_orders') || '{}');
        const order = appState.orders[activeInvoiceId];
        if (order) renderPaymentScreen(order);
      }, 3000);
    }

    // Testfunktion, um zu zeigen, wie eine Unterzahlung (Partial) live greift
    function simulatePaymentTest() {
      if (!activeInvoiceId) return;
      const missing = (appState.orders[activeInvoiceId].totalPrice * 0.4).toFixed(2);
      appState.orders[activeInvoiceId].status = 'PARTIAL';
      appState.orders[activeInvoiceId].remainingPrice = parseFloat(missing);
      saveState();
      renderPaymentScreen(appState.orders[activeInvoiceId]);
    }

    document.getElementById('ticket-form').addEventListener('submit', (e) => {
      e.preventDefault();
      alert("Ticket submitted successfully!");
    });

    document.getElementById('admin-login-form').addEventListener('submit', (e) => {
      e.preventDefault();
      const user = document.getElementById('admin-user').value;
      const pass = document.getElementById('admin-pass').value;
      if (user === 'xaver' && pass === '2011') {
        showView('admin-dashboard');
      } else {
        alert("Falsche Zugangsdaten!");
      }
    });

    function renderAdminDashboard() {
      appState.orders = JSON.parse(localStorage.getItem('store_orders') || '{}');
      const tbody = document.getElementById('admin-orders-table');
      tbody.innerHTML = '';
      Object.values(appState.orders).forEach(ord => {
        const tr = document.createElement('tr');
        tr.innerHTML = `
          <td>${ord.invoiceId}</td>
          <td>${ord.email}</td>
          <td>${ord.product}</td>
          <td>$${ord.totalPrice.toFixed(2)}</td>
          <td><b>${ord.status}</b></td>
          <td>$${ord.remainingPrice.toFixed(2)}</td>
          <td>
            <button onclick="setOrderStatus('${ord.invoiceId}', 'PAID', 0)">Paid</button>
            <button onclick="setOrderStatus('${ord.invoiceId}', 'PARTIAL', 2.50)">Underpaid</button>
          </td>
        `;
        tbody.appendChild(tr);
      });
    }

    function setOrderStatus(invoiceId, status, remaining) {
      appState.orders[invoiceId].status = status;
      appState.orders[invoiceId].remainingPrice = remaining;
      saveState();
      renderAdminDashboard();
    }
